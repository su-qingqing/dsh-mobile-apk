## 1. 端侧可行性判定表（ARM64 Android + Termux / node24 / py3.14 / clang）

### 1.0 一句话结论（先给答案）

> **端侧能干"确定性图形活"（程序化几何、glTF/GLB 组装、2D 合成/抠图后处理/格式转换/缩略图/预览页），干不了"生成式大模型推理"（SD/Flux/3D 生成）。生成必须云端。**

理由三条（硬约束，不是保守）：
1. **内存**：生成式模型最小可用档 SD1.5-q4 GGUF ≈ 700 MB–1 GB 常驻；SDXL ≈ 3–4 GB；**Flux-dev q4 ≈ 6–8 GB** ⇒ 直接越过 Android 单进程可用内存与 Termux 的分配上限。
2. **算力**：8 Gen3（SM8650 / Adreno 750）的 GPU 算力只够跑小模型；Termux 拿系统 Vulkan ICD（`/system/lib64/libvulkan.so`）**成功率不稳定**，且 Adreno 闭源驱动在 Termux 用户态常拿不到。
3. **进程存活**：Android 12+ phantom-process / 内存回收会在后台长任务上杀进程；出图 30–90 s、3D 数分钟，正好是高危窗口。

### 1.1 3D 端侧

| 路径 | ARM64+Termux 可行性 | 依据 / 备注 | 结论 |
|---|---|---|---|
| **three.js 程序化建模 + glTF 直出**（npm，纯 JS） | ✅ **强推荐** | node v24.18.0 实测可用；几何/材质/节点全部 JS 计算，零原生依赖；自己写 glTF JSON + bin buffer 最稳（`GLTFExporter` 在 node 下需 DOM polyfill，绕开） | **主路：端侧程序化 3D** |
| **@gltf-transform/core + functions**（npm，纯 JS/wasm） | ✅ 推荐 | 校验/合并/去重/Draco+Meshopt 压缩/KTX2 贴图，官方 wasm 在 node 下可用 | **主路：GLB 后处理** |

### 6.1 `工具/多模态/件.mjs` —— 公共层（从 rmbg.mjs 抽出来的那层）

```js
#!/usr/bin/env node
// 件.mjs · 多模态公共层：取钥（不回显）/ 落盘（只回一行）/ 记账 / 熔断
// 红线：① 全文不出现任何 key 字面量 ② stdout 只回一行 ③ 大件永不进上下文
import fs from 'node:fs'
import path from 'node:path'
import crypto from 'node:crypto'

const W = '/storage/emulated/0/3'
const 产 = path.join(W, '产')
const 账 = path.join(产, '多模态账.jsonl')
const 凭据 = (process.env.DSH_HOME || '/data/user/0/com.dsharnessmobile.shell/files/home/.dsh')
  + '/.credentials.yaml'

// ── 取钥：只从 env / 凭据文件读，永不打印 ──────────────────────────────
export function 取钥(名) {
  if (process.env[名]) return process.env[名]
  try {
    for (const l of fs.readFileSync(凭据, 'utf8').split('\n')) {
      const m = l.match(new RegExp('^\\s*' + 名 + ':\\s*(\\S+)'))
      if (m) return m[1]
    }
  } catch {}
  return null
}
export const 钥指纹 = (k) => !k ? '未配置'
  : `${k.slice(0, 3)}…${crypto.createHash('sha256').update(k).digest('hex').slice(0, 8)}`

// ── 熔断（fail-closed）───────────────────────────────────────────────
const 上限 = { 单件: 5, 单窗: 5, 单日: 20 }
export function 查花费(范围) {
  if (!fs.existsSync(账)) return 0
  const 今 = new Date(Date.now() + 8 * 3600e3).toISOString().slice(0, 10)
  return fs.readFileSync(账, 'utf8').split('\n').filter(Boolean).map((l) => { try { return JSON.parse(l) } catch { return null } })
    .filter((r) => r && (范围 === '单日' ? r.at.slice(0, 10) === 今 : true))
    .reduce((a, r) => a + (r.花费 || 0), 0)
}
export function 闸(估) {
  const 已 = { 日: 查花费('单日') }
  if (估 > 上限.单件) return `拒：单件估 ¥${估} > 上限 ¥${上限.单件}`
  if (已.日 + 估 > 上限.单日) return `拒：今日已 ¥${已.日.toFixed(2)} + 估 ¥${估} > 日上限 ¥${上限.单日}`
  return null
}

// ── 落盘：大件写盘，只回一行 ──────────────────────────────────────────
export function 落件({ 类, slug, 后缀, 二进制, meta }) {
  const id = crypto.randomBytes(4).toString('hex')
  const 戳 = new Date(Date.now() + 8 * 3600e3).toISOString().slice(0, 10).replace(/-/g, '')
  const 名 = `${戳}-${slug}-${id}`
  const 目录 = path.join(产, 类 === '3d' ? '3d' : '图', 类 === '3d' ? 名 : '')
  fs.mkdirSync(目录, { recursive: true })
  const 件 = path.join(目录, 类 === '3d' ? `model.${后缀}` : `${名}.${后缀}`)
  fs.writeFileSync(件, 二进制)
  fs.writeFileSync(件.replace(/\.[^.]+$/, '.json'), JSON.stringify({ id, ...meta }, null, 1))
  if (meta.花费 != null) {
    fs.appendFileSync(账, JSON.stringify({
      at: new Date(Date.now() + 8 * 3600e3).toISOString(), 类, id,
      模型: meta.模型, 花费: meta.花费, 件: 件.replace(W, ''),
      prompt_sha: meta.prompt ? crypto.createHash('sha256').update(meta.prompt).digest('hex').slice(0, 12) : null,
    }) + '\n')
  }
  return { 件, id, 名 }
}
// ★ 调用方只许打印这一行：
export const 一行 = (r, meta) =>
  `${r.件.replace(W + '/', '')} · ${meta.尺寸 || ''} · ${(r.字节 / 1048576).toFixed(2)}MB · ¥${meta.花费 ?? '?'} · ${meta.耗时}s · 预览 产/页/${r.id}.html`
```

### 6.5 端侧程序化建模骨架（**零依赖，纯 node，一定能跑**）

当老板要的是"参数化的确定性模型"（盒子/齿轮/支架/建筑体块），**别花 3.5 元调云端**，端侧直出 glTF：

```js
#!/usr/bin/env node
// 程序化.mjs · 纯 node 直出 glTF 2.0（无 npm 依赖、无 GPU、离线可跑）
// 用法：node 工具/多模态/程序化.mjs --型 盒 --宽 2 --高 1 --深 1 --出 /storage/emulated/0/3/产/3d/box/box.gltf
import fs from 'node:fs'
import path from 'node:path'

// ① 造几何：一个轴对齐盒子（24 顶点 / 36 索引，带法线）
function 盒(w, h, d) {
  const P = [], N = [], I = []
  const 面 = [
    [[+1, 0, 0], [[w, -h, -d], [w, h, -d], [w, h, d], [w, -h, d]]],
    [[-1, 0, 0], [[-w, -h, d], [-w, h, d], [-w, h, -d], [-w, -h, -d]]],
    [[0, +1, 0], [[-w, h, d], [w, h, d], [w, h, -d], [-w, h, -d]]],
    [[0, -1, 0], [[-w, -h, -d], [w, -h, -d], [w, -h, d], [-w, -h, d]]],
    [[0, 0, +1], [[-w, -h, d], [w, -h, d], [w, h, d], [-w, h, d]]],
    [[0, 0, -1], [[w, -h, -d], [-w, -h, -d], [-w, h, -d], [w, h, -d]]],
  ]
  for (const [n, vs] of 面) {
    const b = P.length / 3
    for (const v of vs) { P.push(...v); N.push(...n) }
    I.push(b, b + 1, b + 2, b, b + 2, b + 3)
  }
  return { P: new Float32Array(P), N: new Float32Array(N), I: new Uint16Array(I) }
}

// ② 打包成 glTF 2.0（.gltf JSON + 外挂 .bin，或内嵌 base64 成单文件）
const g = 盒(1, 0.5, 0.5)
const bin = Buffer.concat([Buffer.from(g.P.buffer), Buffer.from(g.N.buffer), Buffer.from(g.I.buffer)])
const gltf = {
  asset: { version: '2.0', generator: 'dsh-程序化' },
  scenes: [{ nodes: [0] }], scene: 0, nodes: [{ mesh: 0, name: '盒' }],
  meshes: [{ primitives: [{ attributes: { POSITION: 0, NORMAL: 1 }, indices: 2, material: 0 }] }],
  materials: [{ pbrMetallicRoughness: { baseColorFactor: [0.4, 0.7, 0.9, 1], metallicFactor: 0.1, roughnessFactor: 0.6 } }],
  accessors: [
    { bufferView: 0, componentType: 5126, count: g.P.length / 3, type: 'VEC3', min: [-1, -0.5, -0.5], max: [1, 0.5, 0.5] },
    { bufferView: 1, componentType: 5126, count: g.N.length / 3, type: 'VEC3' },
    { bufferView: 2, componentType: 5123, count: g.I.length, type: 'SCALAR' },
  ],
  bufferViews: [
    { buffer: 0, byteOffset: 0, byteLength: g.P.byteLength, target: 34962 },
    { buffer: 0, byteOffset: g.P.byteLength, byteLength: g.N.byteLength, target: 34962 },
    { buffer: 0, byteOffset: g.P.byteLength + g.N.byteLength, byteLength: g.I.byteLength, target: 34963 },
  ],
  buffers: [{ byteLength: bin.length, uri: 'data:application/octet-stream;base64,' + bin.toString('base64') }],
}
const 出 = process.argv.includes('--出') ? process.argv[process.argv.indexOf('--出') + 1] : '/storage/emulated/0/3/产/3d/box/box.gltf'
fs.mkdirSync(path.dirname(出), { recursive: true })
fs.writeFileSync(出, JSON.stringify(gltf))
console.log(`${出} · ${(bin.length / 1024).toFixed(1)}KB · 零依赖 · 离线`)
```
